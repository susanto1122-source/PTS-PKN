Integrasi Google Spreadsheet

Simpan otomatis nama, kelas, absen, nilai, dan jawaban siswa

URL ini otomatis tersimpan di peramban (browser) Anda.

Cara Menghubungkan Google Sheet (Hanya 2 Menit):

  1. Buka sheets.new untuk membuat Google Spreadsheet baru.
  2. Klik menu Extensions (Ekstensi) → Apps Script.
  3. Hapus seluruh kode yang ada di Apps Script, lalu salin & tempel kode di bawah ini.
  4. Klik tombol Deploy (Terapkan) di kanan atas → pilih New deployment (Penerapan baru).
  5. Pilih jenis Web app (Aplikasi Web):
    • Execute as: Me (Email akun Anda)
    • Who has access: Anyone (Siapa saja)
  6. Klik Deploy, izinkan akses akun (Authorize access), lalu salin URL yang berakhiran /exec ke kolom di atas.
Kode Google Apps Script: 
function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    
    // Buat baris judul (header) otomatis jika sheet masih kosong
    if (sheet.getLastRow() === 0) {
      var headers = [
        "Waktu Selesai",
        "Nama Siswa",
        "Kelas",
        "No. Absen",
        "Nilai Akhir",
        "Jumlah Benar",
        "Jumlah Salah",
        "Predikat"
      ];
      
      // Tambahkan kolom jawaban per nomor (Soal 1 s/d Soal 25)
      for (var i = 1; i <= 25; i++) {
        headers.push("Soal " + i);
      }
      headers.push("Ringkasan Analisis");

      sheet.appendRow(headers);
      sheet.getRange(1, 1, 1, headers.length)
        .setFontWeight("bold")
        .setBackground("#1b5387")
        .setFontColor("#ffffff");
    }
    
    // Susun baris data identitas dan nilai siswa
    var rowData = [
      data.timestamp,
      data.nama,
      data.kelas,
      data.absen,
      data.nilai,
      data.benar,
      data.salah,
      data.predikat
    ];

    // Masukkan jawaban siswa untuk masing-masing butir soal (1 s/d 25)
    if (data.jawabanTiapNomor && Array.isArray(data.jawabanTiapNomor)) {
      for (var j = 0; j < data.jawabanTiapNomor.length; j++) {
        rowData.push(data.jawabanTiapNomor[j]);
      }
    }

    // Tambahkan rekap ringkasan detail jawaban di kolom terakhir
    rowData.push(data.detailJawaban || "");
    
    // Tulis baris baru ke sheet
    sheet.appendRow(rowData);
    
    return ContentService.createTextOutput(JSON.stringify({ status: "success" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
